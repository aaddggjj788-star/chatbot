function testSpecialProcess(processName, userText) {
  const text = String(userText || '');

  if (processName === 'saveNickname') {
    const result = extractNickname([text]);

    return {
      processName,
      matched: Boolean(result?.nickname),
      nickname: result?.nickname || '',
      rawResult: result || null
    };
  }

  if (processName === 'saveMemo1') {
    return {
      processName,
      matched: Boolean(text.trim()),
      memoText: text.trim()
    };
  }

  return {
    processName,
    matched: false,
    unsupported: true
  };
}

async function testExcludedReply({
  kid,
  latestComment,
  kanteishiText,
  userText,
  expectedReplyWords = [],
  unmatchedWords = []
}) {
  const item = {
    index: 0,
    uid: 'TEST',
    kid: String(kid || ''),
    receivedAt: '',
    userText: String(userText || ''),
    latestComment: String(latestComment || ''),
    expectedReplyWords:
      Array.isArray(expectedReplyWords) ? expectedReplyWords : [],
    unmatchedWords:
      Array.isArray(unmatchedWords) ? unmatchedWords : [],
    reason: '自動返信対象外: 送り返す言葉が一部不一致'
  };

  const ai = await classifyReturnWordMismatchWithAI(item);

  if (ai.action === 'skip') {
    return {
      decision: ai,
      action: ai.action,
      replyDraft: '',
      commands: []
    };
  }

  const aiContext =
    loadReplyAiContext(item.kid, item.latestComment);

  if (!aiContext.baseProfile) {
    throw new Error(
      `共通AIプロフィール reply-ai-profiles/${item.kid}.json が見つかりません`
    );
  }

  if (!aiContext.phaseProfile) {
    throw new Error(
      '日程別AIプロフィールが見つかりません ' +
      `(requested=${aiContext.requestedPhaseKey || '不明'})`
    );
  }

  const generationInstruction =
    buildSkippedGenerationInstruction(
      item,
      ai,
      item.reason
    );

  const replyDraft =
    await generateProfiledReply({
      aiContext,
      kanteishiText: String(kanteishiText || ''),
      userText: item.userText,
      instruction: generationInstruction
    });

  let commands = [];

  if (ai.action === 'insert_next') {
    commands = [
      '対象外ID:TEST 次行照会',
      `差し込み#${replyDraft}`
    ];
  } else if (ai.action === 'replace_previous') {
    commands = [
      '対象外ID:TEST 次行照会',
      `差し替え前文#${replyDraft}`
    ];
  } else if (ai.action === 'reply_and_resume') {
    commands = [
      `対象外ID:TEST ${replyDraft}`
    ];
  }

  return {
    decision: ai,
    action: ai.action,
    replyDraft,
    commands,
    resolvedPhase: aiContext.phaseKey || null,
    requestedPhase: aiContext.requestedPhaseKey || null
  };
}
